    call salto

    mov [ebx + edi], al     ; se coloca el % al final de la cadena

    call salto          ; se imprime un salto de linea

    mov ebx, cad            ; se manda la direccion de la cadena a ebx

    mov [ebx + edi], al     ; se coloca el % al final de la cadena

    ; imprime la cadena actualizada
    mov ebx, cad2
    call puts

    ; SYS_EXIT
    mov eax, 1
    mov ebx, 0
    int 80h

; ============================================================================
    capturar_og:
    pushad
    capturar_og_loop:
        call getche         ; se captura el caracter y se almacena el al
        cmp al, '*'         ; se compara con el indicador de fin de cadena
        je fin             ; si al == *, se sale del ciclo capturar

        cmp al, 13          ; si al == ENTER, no se captura
        je capturar_og_loop

        cmp al, 10          ; si al == NUEVA_LINEA, no se captura
        je capturar_og_loop
        
        mov [ebx + edi], al ; se coloca el caracter capturado en ebx + indice (edi)
        inc edi             ; se incrementa el indice
        jmp capturar_og_loop        ; se reinicia el ciclo

    ; imprimir cadena capturada
fin:
    popad
    ret
    
capturar:
    pushad              ; Guarda todos los registros generales
    mov edi, 0          ; Inicializa el índice en 0
    capturar_loop:
        call getche         ; Captura el carácter y lo guarda en AL

        cmp al, '*'         ; Verifica si es el fin de la cadena
        je fin_capturar     ; Si es '*', termina el procedimiento

        cmp al, 13          ; Verifica si es ENTER
        je capturar_loop    ; Si es ENTER, ignora y repite

        cmp al, 10          ; Verifica si es NUEVA_LINEA
        je capturar_loop    ; Si es NUEVA_LINEA, ignora y repite

        mov [ebx + edi], al ; Guarda el carácter en el buffer
        inc edi             ; Incrementa el índice
        jmp capturar_loop   ; Repite el ciclo
    fin_capturar:
    popad               ; Restaura los registros
    ret                 ; Retorna al llamador

mayusculas:
    pushad
    mov edi, 0
    change_mayus:
        cmp byte[ebx + edi], '%'
        je fin_change

        sub byte[ebx + edi], 32
        inc edi
        jmp change_mayus
    fin_change:
    popad
    ret

invertir:
    pushad
    mov esi, 0
    mov edi, 0

    
    ciclo_invertir:
        cmp edi, 0
        je fin_invertir

        mov eax, [ebx + edi]
        mov [ebx + esi], eax
        dec edi
        inc esi
        
        jmp ciclo_invertir
    fin_invertir:
    popad
    ret

newputs:
    pushad
    prnt:
        mov al, [ebx + esi]     ; se manda a al el caracter incial
        cmp al, '%'             ; se compara al con % 
        je finputs              ; si es verdadero, se sale del ciclo

        call putchar            ; se imprime al caracter actual
        inc esi                 ; se incrementa el indice
        jmp prnt                ; se reincia el cilo de impresion
    finputs:
    popad
    ret

salto:
    pushad
    mov al, 13
    call putchar

    mov al, 10
    call putchar
    popad
    ret
